        let record = self.resolve(call, request.handle)?;
--- new
        let mut instance = call.instance;
        if request.forged_session.is_some() {
            let live = self
                .registry
                .get(&call.instance)
                .map(|record| record.identity.clone());
            if let Some(live) = live {
                for (id, record) in &self.registry {
                    if *id != call.instance && record.identity == live {
                        instance = *id;
                        break;
                    }
                }
            }
        }
        let record = self.resolve(Call { instance }, request.handle)?;
--- end
