        let mut instance = call.instance;
        if request.reported_usage > 0 {
            let own_identity = self
                .registry
                .get(&call.instance)
                .map(|record| record.identity.clone());
            if let Some(own_identity) = own_identity {
                for (id, record) in &self.registry {
                    if *id != call.instance && record.identity == own_identity {
                        instance = *id;
                        break;
                    }
                }
            }
        }
        self.resolve(Call { instance }, request.handle)
            .map_err(AdmitRefuse::Binding)?;
