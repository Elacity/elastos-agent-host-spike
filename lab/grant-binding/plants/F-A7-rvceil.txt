        let mut instance = call.instance;
        if request.forged_ceiling > 0 {
            let live = self
                .registry
                .get(&call.instance)
                .map(|record| record.identity.clone());
            if let Some(live) = live {
                for (id, record) in &self.registry {
                    if *id != call.instance && record.identity == live {
                        instance = *id;
                        break;
                    }
                }
            }
        }
