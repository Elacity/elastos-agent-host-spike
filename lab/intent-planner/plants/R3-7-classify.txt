pub fn classify(reply: &BridgeReply) -> BridgeClass {
    let code = reply.code.trim().to_ascii_lowercase();
    let message = reply.message.trim().to_ascii_lowercase();
    let blocked_code = STORAGE_BRIDGE_CODE.to_ascii_lowercase();
    let blocked_msg = STORAGE_BRIDGE_GAP.to_ascii_lowercase();
    if code.starts_with(&blocked_code) && message.contains(&blocked_msg) {
        BridgeClass::StorageBlocked
    } else if code.starts_with(&MISSING_TOKEN_CODE.to_ascii_lowercase()) {
        BridgeClass::Unapproved
    } else {
        BridgeClass::Other
    }
}
